    23→    23→/**
    24→    24→ * ============================================================================
    25→    25→ *  CashService — single source of truth for money in Hydra.
    26→    26→ * ============================================================================
    27→    27→ *
    28→    28→ *  DESIGN (Tunisian BTP reality, IntegraSys ERP pattern):
    29→    29→ *
    30→    30→ *   - cash_pot (caisse) = either COMPANY-wide ("Caisse centrale") or PROJECT-scoped
    31→    31→ *     ("Caisse chantier"). Default currency: TND.
    32→    32→ *
    33→    33→ *   - cash_movement = every money event. Lifecycle: DRAFT → POSTED → VOID.
    34→    34→ *     - DRAFT can be edited (PUT) — used when a payment is being prepared but
    35→    35→ *       not yet executed.
    36→    36→ *     - POSTED is IMMUTABLE. To correct a mistake, VOID it and create a new
    37→    37→ *       movement. This is the SYSCOHADA / accrual accounting pattern: the
    38→    38→ *       audit trail is never rewritten.
    39→    39→ *     - VOID is excluded from balance computation but the row is preserved
    40→    40→ *       for traceability.
    41→    41→ *
    42→    42→ *   - balance(pot) is ALWAYS computed as SUM(POSTED movements affecting the pot):
    43→    43→ *       IN: +amount to to_pot
    44→    44→ *       OUT: -amount from from_pot
    45→    45→ *       TRANSFER: -amount from from_pot, +amount to to_pot
    46→    46→ *     Never stored as a mutable field.
    47→    47→ *
    48→    48→ *   - Negative balance guard: when posting an OUT or TRANSFER, if the resulting
    49→    49→ *     from_pot balance would go negative, the post is rejected. This is the
    50→    50→ *     "no overdraft" safety default (Phase 1: ON).
    51→    51→ *
    52→    52→ *  SECURITY:
    53→    53→ *   - Anyone in the org can list pots + movements (read-only visibility for owner).
    54→    54→ *   - Create/update/post/void require: is_owner OR FINANCE business role OR
    55→    55→ *     project membership (for project-scoped pots/movements).
    56→    56→ *   - All org-isolated: queries always filter by caller's org_id.
    57→    57→ *
    58→    58→ * ============================================================================
    59→    59→ *  PHASE 2 — wiring the Process engine to auto-post:
    60→    60→ * ============================================================================
    61→    61→ *
    62→    62→ *  When a PAYMENT_APPROVAL process completes (the gérant clicks "Valider" on
    63→    63→ *  the APPROVE step), ProcessService.completeStep() should call:
    64→    64→ *
    65→    65→ *      cashService.onProcessCompletedPostMovement(movementId);
    66→    66→ *
    67→    67→ *  The movement must already exist as DRAFT with process_instance_id set to
    68→    68→ *  the running instance. To wire this:
    69→    69→ *
    70→    70→ *  1. In ProcessService.start(): after creating the instance, look up the
    71→    71→ *     pending cash_movement by `subject_id` (= the movement id) and set its
    72→    72→ *     process_instance_id. OR have the frontend create the movement DRAFT
    73→    73→ *     first, then start the process with subjectId = movementId.
    74→    74→ *
    75→    75→ *  2. In ProcessService.completeStep(): when the LAST step is completed
    76→    76→ *     (process → COMPLETED), call:
    77→    77→ *
    78→    78→ *         cashService.onProcessCompletedPostMovement(movementId);
    79→    79→ *
    80→    80→ *     This flips the movement DRAFT → POSTED, sets posted_at/by, and the
    81→    81→ *     balance is now affected. If the negative-balance guard fails, the
    82→    82→ *     movement stays DRAFT and the caller gets a BusinessRuleException
    83→    83→ *     (process completion should be rolled back).
    84→    84→ *
    85→    85→ *  3. Phase 2 might also add a `cash_movement.process_status` field to show
    86→    86→ *     "En cours d'approbation" badge in the UI.
    87→    87→ *
    88→    88→ *  The stub method below is the entry point — currently a no-op, marked
    89→    89→ *  @Transactional so Phase 2 can call it from inside ProcessService without
    90→    90→ *  an extra DB roundtrip wrapper.
    91→    91→ *
    92→    92→ * ============================================================================
    93→    93→ */

   149→   149→    /**
   150→   150→     * Require: is_owner OR FINANCE business role OR project member (for project-scoped actions).
   151→   151→     * The "action" param is just for the error message.
   152→   152→     */
   153→   153→    private void requireCanWrite(AppUser me, UUID projectId, String action) {
   154→   154→        if (Boolean.TRUE.equals(me.getIsOwner())) return;
   155→   155→        // Check business role: FINANCE, ACHAT, CHEF_PROJET, DIRECTION
   156→   156→        Integer cnt = jdbc.queryForObject("""
   157→   157→                select count(*) from user_role ur
   158→   158→                join role r on r.id = ur.role_id
   159→   159→                where ur.user_id = ? and r.code in ('FINANCE','OWNER','DIRECTION','ACHAT','CHEF_PROJET')
   160→   160→                """, Integer.class, me.getId());
   161→   161→        if (cnt != null && cnt > 0) return;
   162→   162→        // Project member fallback for project-scoped actions
   163→   163→        if (projectId != null) {
   164→   164→            Integer pm = jdbc.queryForObject(
   165→   165→                    "select count(*) from project_member where project_id = ? and user_id = ?",
   166→   166→                    Integer.class, projectId, me.getId());
   167→   167→            if (pm != null && pm > 0) return;
   168→   168→        }
   169→   169→        throw new BusinessRuleException("FORBIDDEN",
   170→   170→                "Action non autorisée : " + action + " nécessite OWNER/FINANCE/CHEF_PROJET ou appartenance au chantier");
   171→   171→    }
   172→   172→
   173→   173→    /** Stricter check for POST and VOID: is_owner OR FINANCE business role. */
   174→   174→    private void requireCanPostOrVoid(AppUser me) {
   175→   175→        if (Boolean.TRUE.equals(me.getIsOwner())) return;
   176→   176→        Integer cnt = jdbc.queryForObject("""
   177→   177→                select count(*) from user_role ur
   178→   178→                join role r on r.id = ur.role_id
   179→   179→                where ur.user_id = ? and r.code in ('FINANCE','OWNER','DIRECTION')
   180→   180→                """, Integer.class, me.getId());
   181→   181→        if (cnt == null || cnt == 0) {
   182→   182→            throw new BusinessRuleException("FORBIDDEN",
   183→   183→                    "Comptabilisation (POST/VOID) nécessite OWNER/FINANCE/DIRECTION");
   184→   184→        }
   185→   185→    }

   203→   203→    /** List pots visible to caller: COMPANY pots + PROJECT pots for projects they can access. */
   204→   204→    public List<PotResponse> listPots(Authentication auth) {
   205→   205→        AppUser me = currentUser(auth);
   206→   206→
   207→   207→        // COMPANY pots — visible to anyone in the org
   208→   208→        List<PotResponse> result = new ArrayList<>(jdbc.queryForList("""
   209→   209→                select id, org_id, type, project_id, name, currency, active, created_at
   210→   210→                from cash_pot
   211→   211→                where org_id = ? and type = 'COMPANY'
   212→   212→                order by active desc, name asc
   213→   213→                """, me.getOrgId())
   214→   214→                .stream().map(CashService::toPotResponse).toList());
   215→   215→
   216→   216→        // PROJECT pots — only for projects the caller is a member of (or owner)
   217→   217→        if (Boolean.TRUE.equals(me.getIsOwner())) {
   218→   218→            result.addAll(jdbc.queryForList("""
   219→   219→                    select id, org_id, type, project_id, name, currency, active, created_at
   220→   220→                    from cash_pot
   221→   221→                    where org_id = ? and type = 'PROJECT'
   222→   222→                    order by active desc, name asc
   223→   223→                    """, me.getOrgId())
   224→   224→                    .stream().map(CashService::toPotResponse).toList());
   225→   225→        } else {
   226→   226→            result.addAll(jdbc.queryForList("""
   227→   227→                    select cp.id, cp.org_id, cp.type, cp.project_id, cp.name, cp.currency, cp.active, cp.created_at
   228→   228→                    from cash_pot cp
   229→   229→                    where cp.org_id = ? and cp.type = 'PROJECT'
   230→   230→                      and exists (
   231→   231→                        select 1 from project_member pm where pm.project_id = cp.project_id and pm.user_id = ?
   232→   232→                      )
   233→   233→                    order by cp.active desc, cp.name asc
   234→   234→                    """, me.getOrgId(), me.getId())
   235→   235→                    .stream().map(CashService::toPotResponse).toList());
   236→   236→        }
   237→   237→
   238→   238→        return result;
   239→   239→    }

   241→   241→    /**
   242→   242→     * Ensure the org has its "Caisse centrale" (COMPANY pot). Idempotent —
   243→   243→     * if it already exists (and is active), returns it.
   244→   244→     */
   245→   245→    @Transactional
   246→   246→    public PotResponse ensureCompanyPot(Authentication auth) {
   247→   247→        AppUser me = currentUser(auth);
   248→   248→        requireCanWrite(me, null, "ensure-company-pot");
   249→   249→
   250→   250→        // Try to fetch existing active company pot
   251→   251→        List<Map<String, Object>> rows = jdbc.queryForList("""
   252→   252→                select id, org_id, type, project_id, name, currency, active, created_at
   253→   253→                from cash_pot
   254→   254→                where org_id = ? and type = 'COMPANY' and active = true
   255→   255→                limit 1
   256→   256→                """, me.getOrgId());
   257→   257→        if (!rows.isEmpty()) {
   258→   258→            return toPotResponse(rows.get(0));
   259→   259→        }
   260→   260→
   261→   261→        // Also check for an INACTIVE one to reactivate
   262→   262→        List<Map<String, Object>> inactive = jdbc.queryForList(
   263→   263→                "select id from cash_pot where org_id = ? and type = 'COMPANY' and active = false limit 1",
   264→   264→                me.getOrgId());
   265→   265→        if (!inactive.isEmpty()) {
   266→   266→            UUID id = (UUID) inactive.get(0).get("id");
   267→   267→            jdbc.update("update cash_pot set active = true where id = ?", id);
   268→   268→            return getPot(auth, id);
   269→   269→        }
   270→   270→
   271→   271→        // Create a fresh one
   272→   272→        UUID id = UUID.randomUUID();
   273→   273→        jdbc.update("""
   274→   274→                insert into cash_pot (id, org_id, type, project_id, name, currency, active, created_at)
   275→   275→                values (?, ?, 'COMPANY', null, 'Caisse centrale', 'TND', true, now())
   276→   276→                """, id, me.getOrgId());
   277→   277→
   278→   278→        audit.log("CASH_POT_CREATED", "cash_pot", id,
   279→   279→                "{\"type\":\"COMPANY\",\"name\":\"Caisse centrale\"}");
   280→   280→
   281→   281→        return getPot(auth, id);
   282→   282→    }

   295→   295→                select id, org_id, type, project_id, name, currency, active, created_at
   296→   296→                from cash_pot
   297→   297→                where org_id = ? and project_id = ? and type = 'PROJECT'

   301→   301→            // Reactivate if needed
   302→   302→            Map<String, Object> row = rows.get(0);

   308→   308→
   309→   309→        // Get project name for the pot name

   317→   317→
   318→   318→        UUID id = UUID.randomUUID();
   319→   319→        jdbc.update("""
   320→   320→                insert into cash_pot (id, org_id, type, project_id, name, currency, active, created_at)
   321→   321→                values (?, ?, 'PROJECT', ?, ?, 'TND', true, now())
   322→   322→                """, id, me.getOrgId(), projectId, potName);
   323→   323→
   324→   324→        audit.log("CASH_POT_CREATED", "cash_pot", id,
   325→   325→                "{\"type\":\"PROJECT\",\"projectId\":\"" + projectId + "\",\"name\":\"" + jsonEscape(potName) + "\"}");
   326→   326→
   327→   327→        return getPot(auth, id);
   328→   328→    }

   333→   333→        List<Map<String, Object>> rows = jdbc.queryForList("""

   348→   348→    /** Balance for a pot — SUM of POSTED movements. */
   349→   349→    public BalanceResponse balance(Authentication auth, UUID potId) {
   350→   350→        AppUser me = currentUser(auth);
   351→   351→        PotResponse pot = getPot(auth, potId); // validates access
   352→   352→
   353→   353→        BigDecimal bal = computeBalance(pot.id());
   354→   354→        return new BalanceResponse(pot.id(), bal, pot.currency());
   355→   355→    }
   356→   356→
   357→   357→    /**
   358→   358→     * Pure balance computation: SUM(POSTED IN to pot) - SUM(POSTED OUT from pot)
   359→   359→     * + SUM(POSTED TRANSFER to pot) - SUM(POSTED TRANSFER from pot).
   360→   360→     * Visible to the service only — used by post() to enforce the no-overdraft rule.
   361→   361→     */
   362→   362→    private BigDecimal computeBalance(UUID potId) {
   363→   363→        BigDecimal bal = jdbc.queryForObject("""
   364→   364→                select coalesce(sum(amount), 0) from cash_movement
   365→   365→                where status = 'POSTED' and to_pot_id = ? and movement_type in ('IN','TRANSFER')
   366→   366→                """, BigDecimal.class, potId);
   367→   367→        BigDecimal out = jdbc.queryForObject("""
   368→   368→                select coalesce(sum(amount), 0) from cash_movement
   369→   369→                where status = 'POSTED' and from_pot_id = ? and movement_type in ('OUT','TRANSFER')
   370→   370→                """, BigDecimal.class, potId);
   371→   371→        if (bal == null) bal = BigDecimal.ZERO;
   372→   372→        if (out == null) out = BigDecimal.ZERO;
   373→   373→        return bal.subtract(out);
   374→   374→    }

   380→   380→    /**
   381→   381→     * Create a DRAFT movement. Validates type/pots/amount/date + caller has write access.
   382→   382→     * processInstanceId may be set later (Phase 2) when a workflow starts.
   383→   383→     */
   384→   384→    @Transactional
   385→   385→    public MovementResponse createMovement(Authentication auth, CreateMovementRequest req) {
   386→   386→        AppUser me = currentUser(auth);
   387→   387→        validateMovementInput(req.movementType(), req.amount(), req.movementDate(),
   388→   388→                req.fromPotId(), req.toPotId());
   389→   389→
   390→   390→        // Resolve project_id: from request, or from PROJECT pot (when from/to is project pot)
   391→   391→        UUID resolvedProjectId = req.projectId();
   392→   392→        if (resolvedProjectId == null) {
   393→   393→            resolvedProjectId = lookupProjectFromPot(req.fromPotId(), me.getOrgId());
   394→   394→            if (resolvedProjectId == null) {
   395→   395→                resolvedProjectId = lookupProjectFromPot(req.toPotId(), me.getOrgId());
   396→   396→            }
   397→   397→        }
   398→   398→
   399→   399→        requireCanWrite(me, resolvedProjectId, "create-movement");
   400→   400→
   401→   401→        // Validate pot ownership (must be in caller's org)
   402→   402→        validatePotInOrg(req.fromPotId(), me.getOrgId());
   403→   403→        validatePotInOrg(req.toPotId(), me.getOrgId());
   404→   404→        if (resolvedProjectId != null) {
   405→   405→            requireProjectAccess(me, resolvedProjectId);
   406→   406→        }
   407→   407→
   408→   408→        UUID id = UUID.randomUUID();
   409→   409→        jdbc.update("""
   410→   410→                insert into cash_movement
   411→   411→                  (id, org_id, movement_type, amount, currency, movement_date,
   412→   412→                   from_pot_id, to_pot_id, project_id, party_id, label, note,
   413→   413→                   status, process_instance_id, created_by, created_at)
   414→   414→                values (?, ?, ?, ?, 'TND', ?, ?, ?, ?, ?, ?, ?, 'DRAFT', ?, ?, now())
   415→   415→                """, id, me.getOrgId(), normalizeType(req.movementType()), req.amount(),
   416→   416→                Date.valueOf(req.movementDate()),
   417→   417→                req.fromPotId(), req.toPotId(), resolvedProjectId, req.partyId(),
   418→   418→                blankToNull(req.label()), blankToNull(req.note()),
   419→   419→                req.processInstanceId(),

   550→   550→     * POST a DRAFT movement → IMMUTABLE POSTED. Sets posted_at/by.
   551→   551→     * For OUT/TRANSFER: enforces no-negative-balance guard on from_pot.

   554→   554→    public PostResponse postMovement(Authentication auth, UUID id) {
   555→   555→        AppUser me = currentUser(auth);
   556→   556→        requireCanPostOrVoid(me);
   557→   557→
   558→   558→        Map<String, Object> row = fetchMovementForUpdate(id, me.getOrgId());
   559→   559→        String status = (String) row.get("status");
   560→   560→        if (!"DRAFT".equals(status)) {
   561→   561→            throw new BusinessRuleException("MOVEMENT_NOT_DRAFT",
   562→   562→                    "Seul un mouvement DRAFT peut être comptabilisé (POST). Statut actuel: " + status);
   563→   563→        }
   564→   564→
   565→   565→        String type = (String) row.get("movement_type");
   566→   566→        UUID fromPotId = (UUID) row.get("from_pot_id");
   567→   567→        UUID toPotId = (UUID) row.get("to_pot_id");
   568→   568→        BigDecimal amount = (BigDecimal) row.get("amount");
   569→   569→        UUID projectId = (UUID) row.get("project_id");
   570→   570→
   571→   571→        // Negative balance guard for OUT / TRANSFER
   572→   572→        BigDecimal fromPotBalanceAfter = null;
   573→   573→        if ("OUT".equals(type) || "TRANSFER".equals(type)) {
   574→   574→            BigDecimal before = computeBalance(fromPotId);
   575→   575→            BigDecimal after = before.subtract(amount);
   576→   576→            fromPotBalanceAfter = after;
   577→   577→            if (after.signum() < 0) {
   578→   578→                throw new BusinessRuleException("INSUFFICIENT_BALANCE",
   579→   579→                        "Solde insuffisant sur la caisse source (" + before + " < " + amount + " " + row.get("currency") + ")");
   580→   580→            }
   581→   581→        }
   582→   582→
   583→   583→        jdbc.update("""
   584→   584→                update cash_movement set
   585→   585→                  status = 'POSTED', posted_by = ?, posted_at = now()
   586→   586→                where id = ? and org_id = ?
   587→   587→                """, me.getId(), id, me.getOrgId());
   588→   588→
   589→   589→        audit.log("CASH_MOVEMENT_POSTED", "cash_movement", id,
   590→   590→                "{\"type\":\"" + type + "\",\"amount\":" + amount +
   591→   591→                (fromPotId != null ? ",\"fromPot\":\"" + fromPotId + "\"" : "") +
   592→   592→                (toPotId != null ? ",\"toPot\":\"" + toPotId + "\"" : "") +
   593→   593→                "}");
   594→   594→
   595→   595→        return new PostResponse(id, "POSTED", fromPotBalanceAfter);
   596→   596→    }

   670→   670→    private void validateMovementInput(String type, BigDecimal amount, LocalDate date,
   671→   671→                                        UUID fromPotId, UUID toPotId) {
   672→   672→        String t = normalizeType(type);
   673→   673→        if (amount == null || amount.signum() <= 0) {
   674→   674→            throw new BadRequestException("INVALID_AMOUNT",
   675→   675→                    "Le montant doit être positif");
   676→   676→        }
   677→   677→        if (date == null) {
   678→   678→            throw new BadRequestException("DATE_REQUIRED",
   679→   679→                    "La date du mouvement est obligatoire");
   680→   680→        }
   681→   681→        switch (t) {
   682→   682→            case "IN" -> {
   683→   683→                if (toPotId == null || fromPotId != null) {
   684→   684→                    throw new BadRequestException("INVALID_POTS_FOR_IN",
   685→   685→                            "IN: to_pot_id requis, from_pot_id doit être null");
   686→   686→                }
   687→   687→            }
   688→   688→            case "OUT" -> {
   689→   689→                if (fromPotId == null || toPotId != null) {
   690→   690→                    throw new BadRequestException("INVALID_POTS_FOR_OUT",
   691→   691→                            "OUT: from_pot_id requis, to_pot_id doit être null");
   692→   692→                }
   693→   693→            }
   694→   694→            case "TRANSFER" -> {
   695→   695→                if (fromPotId == null || toPotId == null) {
   696→   696→                    throw new BadRequestException("INVALID_POTS_FOR_TRANSFER",
   697→   697→                            "TRANSFER: from_pot_id et to_pot_id requis");
   698→   698→                }
   699→   699→                if (fromPotId.equals(toPotId)) {
   700→   700→                    throw new BadRequestException("SAME_POT_TRANSFER",
   701→   701→                            "TRANSFER: la caisse source et destination doivent être différentes");
   702→   702→                }
   703→   703→            }
   704→   704→            default -> throw new BadRequestException("INVALID_MOVEMENT_TYPE",
   705→   705→                    "Type invalide: " + type + " (IN/OUT/TRANSFER)");
   706→   706→        }
   707→   707→    }
